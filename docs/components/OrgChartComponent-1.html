  id: string;

export class OrgChartComponent {
  root = input<OrgChartNode | null>(null);
  compact = input(false);

  nodeSelected = output<OrgChartNode>();

  initials(label: string): string {
    return label
      .split(/\s+/)
      .filter(Boolean)
      .slice(0, 2)
      .map(part => part[0]?.toUpperCase())
      .join('');
  }
}
